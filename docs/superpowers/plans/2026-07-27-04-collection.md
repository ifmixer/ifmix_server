# 模块 collection（收藏）实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development 或
> superpowers:executing-plans。步骤用复选框（`- [ ]`）跟踪。

**目标：** 移植旧项目 collection：用户把扫描结果加入默认收藏夹、列表查看、批量删除。模型
1 collection : N collection_item（独立 join 表，预留多命名夹/一扫描进多夹）。附带把 `collected`
标记回填到 antique 的 `ScanDto`（经端口注入，antique 不 import collection）。

**架构：** `CollectionDocument`/`CollectionItemDocument` 均 `BaseAppDocument`（appId 分片 + 软删）。
`CollectionService` 组合持有 collection 的 `CRUDService` + `collectionItemRepo`。归属判定用共享助手
`ownsRow`/`ownerCriteria`（登录 userId OR installId；匿名仅 installId）。`getDefault` get-or-create
默认夹（partial unique 兜底并发）。`listItems` join `collection_item → scan_record`，复用 antique 的
`ScanDto` + presign。实现 antique 的 `CollectionMembership` 端口（collection → antique 单向依赖）。

**技术栈：** Kotlin 2.3.10、Spring Data MongoDB、`TxRunner`（并发回读）。

**环境约束：** 无 Docker → 集成测试 Testcontainers（CI 跑）；归属助手等纯逻辑单元测试可跑。**不要自动 commit。**

```kotlin
package com.ifmix.api.core.common.db

import com.ifmix.api.core.common.http.RequestContext
import org.springframework.data.mongodb.core.query.Criteria

/** 行归属判定：登录 userId 或 installId 命中；匿名仅 installId。 */
fun ownsRow(ctx: RequestContext, userId: String?, installId: String?): Boolean =
    (ctx.userId != null && userId == ctx.userId) || (installId != null && installId == ctx.installId)

/** 归属查询条件：登录时 userId OR installId；匿名仅 installId。 */
fun ownerCriteria(ctx: RequestContext): Criteria {
    val install = Criteria.where("installId").`is`(ctx.installId)
    return if (ctx.userId != null) {
        Criteria().orOperator(Criteria.where("userId").`is`(ctx.userId), install)
    } else {
        install
    }
}
```

```kotlin
    fun insertIfAbsent(ctx: RequestContext, collectionId: String, scanRecordId: String): String {
        val existing = mongo.findOne(
            Query(base(ctx, collectionId).and("scanRecordId").`is`(scanRecordId).and("deletedAt").`is`(null)),
            CollectionItemDocument::class.java,

        if (existing?.id != null) return existing.id!!

            appId = ctx.appId
            this.collectionId = collectionId
            this.scanRecordId = scanRecordId
            createdAt = now
            updatedAt = now

        mongo.insert(doc)
        return doc.id!!

    fun softDeleteByScanIds(ctx: RequestContext, collectionId: String, scanRecordIds: List<String>): Long {
```

- [ ] **步骤 2：补全 `listScanRecords`（无占位）**

```kotlin
fun listScanRecords(ctx: RequestContext, collectionId: String, cursor: String?, limit: Int): Pair<List<ScanRecordDocument>, String?> {
    // 第一步：取该夹未删 item 的 scanRecordId（按 _id desc keyset 分页 item）
    val itemQuery = Query(base(ctx, collectionId).and("deletedAt").`is`(null))
    if (!cursor.isNullOrBlank() && ObjectId.isValid(cursor)) {
        itemQuery.addCriteria(Criteria.where("scanRecordId").lt(ObjectId(cursor)))
    }
    itemQuery.with(org.springframework.data.domain.Sort.by(org.springframework.data.domain.Sort.Direction.DESC, "scanRecordId"))
    itemQuery.limit(limit + 1)
    val items = mongo.find(itemQuery, CollectionItemDocument::class.java)
    val hasMore = items.size > limit
    val page = if (hasMore) items.subList(0, limit) else items
    val scanIds = page.mapNotNull { it.scanRecordId }.filter { ObjectId.isValid(it) }.map { ObjectId(it) }
    if (scanIds.isEmpty()) return Pair(emptyList(), null)
    // 第二步：批量取 scan_record（appId 分片 + 未删），保持 item 顺序
    val scans = mongo.find(
        Query(Criteria.where("appId").`is`(ctx.appId).and("_id").`in`(scanIds).and("deletedAt").`is`(null)),
        ScanRecordDocument::class.java,
    ).associateBy { it.id }
    val ordered = page.mapNotNull { scans[it.scanRecordId] }
    val nextCursor = if (hasMore) page.last().scanRecordId else null
    return Pair(ordered, nextCursor)
}
```
> 说明：游标用 `scanRecordId`（= scan_record._id，ObjectId 时间有序）做 keyset，与旧项目"scan_record.id
> desc"语义一致。item 表 `scanRecordId` 存 hex 字符串；`lt(ObjectId(cursor))` 需 item 的 scanRecordId
> 以 ObjectId 存。**决策点：** collection_item.scanRecordId 存 ObjectId 还是 hex 字符串——建议存 hex
> （与其他外键一致），则 keyset 比较改为字符串比较或改存 ObjectId。实现时统一（下方任务 6 集成测试验证）。

```kotlin
package com.ifmix.api.core.modules.collection

import com.ifmix.api.core.common.db.CursorQueryInput
import com.ifmix.api.core.common.db.Page
import com.ifmix.api.core.common.db.ownsRow
import com.ifmix.api.core.common.http.ApiError
import com.ifmix.api.core.common.http.ErrorCode
import com.ifmix.api.core.common.http.RequestContext
import com.ifmix.api.core.common.service.CRUDService
import com.ifmix.api.core.modules.antique.ScanRecordDocument

class CollectionService(
    private val collectionCrud: CRUDService<CollectionDocument>,
    private val itemRepo: CollectionItemRepository,
    private val collectionRepo: CollectionRepository,  // 见任务 5：按归属查默认夹
) {
    /** get-or-create 默认夹。 */
    fun getDefault(ctx: RequestContext): CollectionDocument {
        collectionRepo.findDefault(ctx)?.let { return it }
        val doc = CollectionDocument().apply {
            installId = ctx.installId
            userId = ctx.userId
            isDefault = true
        }
        return try {
            collectionCrud.createOne(ctx, doc)
            doc
        } catch (e: org.springframework.dao.DuplicateKeyException) {
            // 并发：partial unique 兜底，回读（走主库由 ctx.readPreference 默认 primaryPreferred 保证）
            collectionRepo.findDefault(ctx) ?: throw e
        }
    }

    private fun resolveCollectionId(ctx: RequestContext, collectionId: String?): String {
        if (collectionId == null) return getDefault(ctx).id!!
        val coll = collectionCrud.findById(ctx, collectionId) ?: throw ApiError(ErrorCode.NOT_FOUND)
        if (!ownsRow(ctx, coll.userId, coll.installId)) throw ApiError(ErrorCode.NOT_FOUND)
        return coll.id!!
    }

    fun addItem(ctx: RequestContext, req: AddItemReq): String {
        val cid = resolveCollectionId(ctx, req.collectionId)
        return itemRepo.insertIfAbsent(ctx, cid, req.scanRecordId!!)
    }

    fun removeItems(ctx: RequestContext, req: RemoveItemsReq): Long {
        val cid = resolveCollectionId(ctx, req.collectionId)
        return itemRepo.softDeleteByScanIds(ctx, cid, req.scanRecordIds!!)
    }

    fun listItems(ctx: RequestContext, req: ListItemsReq): Page<ScanRecordDocument> {
        val cid = resolveCollectionId(ctx, req.collectionId)
        val limit = (req.limit ?: CursorQueryInput.DEFAULT_LIMIT).coerceIn(1, CursorQueryInput.MAX_LIMIT)
        val (scans, next) = itemRepo.listScanRecords(ctx, cid, req.cursor, limit)
        return Page(scans, next, next != null)
    }
}
```

```kotlin
import com.ifmix.api.core.common.http.RequestContext
import com.ifmix.api.core.modules.antique.CollectionMembership
import org.springframework.data.mongodb.core.MongoTemplate
import org.springframework.data.mongodb.core.query.Criteria
import org.springframework.data.mongodb.core.query.Query

class CollectionMembershipImpl(
    private val mongo: MongoTemplate,
    private val collectionRepo: CollectionRepository,
) : CollectionMembership {
    override fun isCollected(ctx: RequestContext, scanRecordId: String): Boolean {
        val cid = collectionRepo.findDefault(ctx)?.id ?: return false
        return mongo.exists(
            Query(
                Criteria.where("appId").`is`(ctx.appId)
                    .and("collectionId").`is`(cid)
                    .and("scanRecordId").`is`(scanRecordId)
                    .and("deletedAt").`is`(null),
            ),
            CollectionItemDocument::class.java,
        )
    }
}
```

```kotlin
@RestController
@RequestMapping("/customer/core")
class CustomerCollectionController(
    private val service: CollectionService,
    private val storage: ObjectStorage,

    private val scanMapper: ScanMapper = Konverter.get()

    @PutMapping("/query/collection/getDefault")

        return GetDefaultRes(c.id, c.isDefault, c.createdAt)

    @PostMapping("/mutation/collection/addItem")
    fun addItem(ctx: RequestContext, @Valid @RequestBody req: AddItemReq): AddItemRes =
        AddItemRes(service.addItem(ctx, req))

    @PostMapping("/mutation/collection/removeItems")
    fun removeItems(ctx: RequestContext, @Valid @RequestBody req: RemoveItemsReq): RemoveItemsRes =
        RemoveItemsRes(service.removeItems(ctx, req))

    @PutMapping("/query/collection/listItems")
    fun listItems(ctx: RequestContext, @RequestBody(required = false) req: ListItemsReq?): Page<ScanDto> {
        val page = service.listItems(ctx, req ?: ListItemsReq())
        return Page(page.items.map { scanMapper.toDto(it).copy(imageUrl = it.imageKey?.let { k -> storage.presignGetUrl(k) }, collected = true) }, page.nextCursor, page.hasMore)
    }
}
```
