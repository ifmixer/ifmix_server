# 模块 iap（内购订阅 + webhooks + 档位联动）实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development 或
> superpowers:executing-plans。步骤用复选框（`- [ ]`）跟踪。

**目标：** 移植旧项目 iap：客户端购买后上送凭证 → 服务端向 Apple/Google 验证并归一化订阅状态 →
upsert 落库 → 返回档位；Apple ASSN v2 / Google RTDN webhook 主动更新续订/取消/退款/过期；把有效
订阅接入 antique 限流档位（有效订阅 → pro）。模块 app 无关，按 appId 从 `app_config` 读商店凭证。

**架构：** `SubscriptionDocument`/`StoreNotificationDocument` 用 `BaseAppDocument`（appId 分片）。
验证/解码抽象为 `PurchaseVerifier`/`NotificationDecoder` 接缝（Apple/Google 各一实现，依赖外部库 +
凭证，凭证就绪前提供骨架 + stub）。`SubscriptionRepo` 提供 `upsert`/`updateByOriginalTxn`/
`findActiveBySubject`。`entitlement.createIapTierResolver` 供 antique 的 `TierResolver` 注入（解耦：
ratelimit 不依赖 iap，容器接线）。

**技术栈：** Kotlin 2.3.10、Spring Data MongoDB、`TxRunner`、Apple `app-store-server-library`（Java）、
Google `google-api-services-androidpublisher`。

**环境约束：** 无 Docker + 无真实商店凭证 → verify/webhook 端到端联调推迟；`tierOf`/状态映射/
`findActiveBySubject`/subject 阶梯等**纯逻辑单元测试可跑**；落库用 Testcontainers 集成测试（CI）。
**不要自动 commit。**

```kotlin
package com.ifmix.api.core.modules.iap

import com.ifmix.api.core.common.ratelimit.Tier
import java.time.Instant

enum class Platform { IOS, ANDROID }

enum class SubStatus { ACTIVE, IN_GRACE, CANCELED, EXPIRED, REVOKED }

/** 归一化订阅状态（验证/解码的统一输出）。 */
data class SubscriptionState(
    val platform: Platform,
    val productId: String,
    val originalTransactionId: String,
    val status: SubStatus,
    val expiresAt: Instant?,
    val autoRenew: Boolean,
    val environment: String,
    val raw: Map<String, Any?> = emptyMap(),
)

fun statusFromExpiry(expiresAt: Instant?, now: Instant = Instant.now()): SubStatus =
    if (expiresAt != null && expiresAt.isBefore(now)) SubStatus.EXPIRED else SubStatus.ACTIVE

fun tierOf(state: SubscriptionState, now: Instant = Instant.now()): Tier {
    val active = state.status == SubStatus.ACTIVE || state.status == SubStatus.IN_GRACE
    val notExpired = state.expiresAt == null || state.expiresAt.isAfter(now)
    return if (active && notExpired) Tier.PRO else Tier.FREE
}
```

```kotlin
package com.ifmix.api.core.modules.iap

import com.ifmix.api.core.common.http.RequestContext
import org.springframework.data.mongodb.core.MongoTemplate
import org.springframework.data.mongodb.core.query.Criteria
import org.springframework.data.mongodb.core.query.Query
import org.springframework.data.mongodb.core.query.Update
import org.springframework.data.mongodb.core.query.isEqualTo
import java.time.Instant

/**
 * 订阅仓储。upsert 按 (appId, platform, originalTransactionId)；updateByOriginalTxn 供 webhook；
 * findActiveBySubject 按 userId OR installId 匹配当前有效订阅（D1 前向兼容）。
 */
class SubscriptionRepo(private val mongo: MongoTemplate) {

    fun upsert(ctx: RequestContext, installId: String?, userId: String?, state: SubscriptionState) {
        val now = Instant.now()
        val platform = state.platform.name.lowercase()
        val query = Query(
            Criteria.where("appId").isEqualTo(ctx.appId)
                .and("platform").isEqualTo(platform)
                .and("originalTransactionId").isEqualTo(state.originalTransactionId),
        )
        val update = Update()
            .setOnInsert("appId", ctx.appId)
            .setOnInsert("platform", platform)
            .setOnInsert("originalTransactionId", state.originalTransactionId)
            .setOnInsert("createdAt", now)
            .setOnInsert("deletedAt", null)
            .set("installId", installId)
            .set("productId", state.productId)
            .set("status", state.status.name.lowercase())
            .set("expiresAt", state.expiresAt)
            .set("autoRenew", state.autoRenew)
            .set("environment", state.environment)
            .set("raw", state.raw)
            .set("updatedAt", now)
        if (userId != null) update.set("userId", userId)  // 已登录购买回填；匿名不覆盖
        mongo.upsert(query, update, SubscriptionDocument::class.java)
    }

    fun updateByOriginalTxn(appId: String, platform: Platform, originalTransactionId: String, state: SubscriptionState) {
        val now = Instant.now()
        val query = Query(
            Criteria.where("appId").isEqualTo(appId)
                .and("platform").isEqualTo(platform.name.lowercase())
                .and("originalTransactionId").isEqualTo(originalTransactionId),
        )
        val update = Update()
            .set("status", state.status.name.lowercase())
            .set("expiresAt", state.expiresAt)
            .set("autoRenew", state.autoRenew)
            .set("raw", state.raw)
            .set("updatedAt", now)
        mongo.updateFirst(query, update, SubscriptionDocument::class.java)
    }

    /** userId + installId 都在时按 OR 匹配（同 appId）；仅其一按其一。返回有效（active/in_grace 未过期）订阅。 */
    fun findActiveBySubject(appId: String, userId: String?, installId: String?): SubscriptionDocument? {
        val subject = when {
            userId != null && installId != null -> Criteria().orOperator(
                Criteria.where("userId").isEqualTo(userId),
                Criteria.where("installId").isEqualTo(installId),
            )
            userId != null -> Criteria.where("userId").isEqualTo(userId)
            installId != null -> Criteria.where("installId").isEqualTo(installId)
            else -> return null
        }
        val query = Query(
            Criteria.where("appId").isEqualTo(appId)
                .andOperator(subject, Criteria.where("status").`in`("active", "in_grace")),
        )
        return mongo.find(query, SubscriptionDocument::class.java)
            .filter { it.expiresAt == null || it.expiresAt!!.isAfter(Instant.now()) }
            .maxByOrNull { it.expiresAt ?: Instant.MAX }
    }
}
```

```kotlin
package com.ifmix.api.core.modules.iap

import com.ifmix.api.core.common.http.ApiError
import com.ifmix.api.core.common.http.ErrorCode
import com.ifmix.api.core.common.http.RequestContext
import com.ifmix.api.core.common.ratelimit.Tier
import com.ifmix.api.core.modules.appconfig.AppConfigRepo

class IapService(
    private val appConfigRepo: AppConfigRepo,
    private val subscriptionRepo: SubscriptionRepo,
    private val appleVerifier: PurchaseVerifier,
    private val googleVerifier: PurchaseVerifier,
) {
    fun verifyPurchase(ctx: RequestContext, req: VerifyReq): VerifyRes {
        if (ctx.installId.isNullOrBlank()) throw ApiError(ErrorCode.INVALID_REQUEST, "x-install-id: required")
        val config = appConfigRepo.getByAppId(ctx.appId) ?: throw ApiError(ErrorCode.APP_CONFIG_MISSING)
        val verifier = if (req.platform == Platform.IOS) appleVerifier else googleVerifier
        val state = try {
            verifier.verify(config, VerifyInput(req.platform!!, req.signedTransaction, req.purchaseToken))
        } catch (e: ApiError) {
            throw e
        } catch (e: Exception) {
            throw ApiError(ErrorCode.IAP_VERIFY_FAILED, e.message ?: "verify failed")
        }
        subscriptionRepo.upsert(ctx, ctx.installId, ctx.userId, state)
        val tier = tierOf(state)
        return VerifyRes(if (tier == Tier.PRO) "pro" else "free", state.status.name.lowercase(), state.expiresAt?.toEpochMilli())
    }
}
```

```kotlin
package com.ifmix.api.core.bff.webhooks

import com.ifmix.api.core.modules.iap.IapService
import org.springframework.http.ResponseEntity
import org.springframework.web.bind.annotation.*

/**
 * 商店 webhook：Apple/Google 不带 x-app-id/x-install-id，故挂在 /webhooks（不经 header 校验拦截器，
 * 见 WebConfig 只拦 /customer 与 /app）。返回裸状态码（非信封）——但 EnvelopeResponseAdvice 会包装，
 * 故返回 ResponseEntity<String> 并借助 advice 对 String 的跳过规则（StringHttpMessageConverter 不包装）。
 */
@RestController
@RequestMapping("/webhooks/iap")
class WebhookController(private val iapService: IapService) {

    @PostMapping("/apple")
    fun apple(@RequestBody rawBody: String): ResponseEntity<String> {
        iapService.handleAppleNotification(rawBody)
        return ResponseEntity.ok("")  // Apple 期望 2xx ack
    }

    @PostMapping("/google")
    fun google(@RequestBody rawBody: String): ResponseEntity<String> {
        iapService.handleGoogleNotification(rawBody)
        return ResponseEntity.noContent().build()  // Google 期望 204
    }
}
```
> **验证点：** `EnvelopeResponseAdvice` 对 `StringHttpMessageConverter` 返回跳过包装（已确认），故
> String 响应体不被包成信封，符合 webhook ack 要求。webhook 失败应返非 2xx 触发商店重投——让异常
> 冒泡到 `GlobalExceptionHandler`（会包信封 + 5xx），Apple 重投可接受。
