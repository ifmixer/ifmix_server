**目标：** 移植旧项目 antique 的"未登录扫描"切片：客户端直传图片到 R2 → 调 `newScan` → 按

`scan_record` → 返回结果；失败退还限额。附带历史查询 `listScans`/`getScanById` 与通用 R2 预签名。

**架构：** 新增基础设施 `common/redis`（Lettuce）、`common/ratelimit`（UTC 日固定窗口）、
`common/storage`（R2/S3 AWS SDK v2 预签名）、`common/http/ClientIpResolver`。`scan_record` 用
`BaseAppDocument`（app 级 + 软删）。`AntiqueService` **组合**持有 `rateLimiter`/`tierResolver`/
`r2`/`scanRunner`/`scanCrud`，编排扫描流程。AI 调用抽象为 `ScanRunner` 接缝，v1 用 stub 打通端到端；
真实 AI（Spring AI / agnes key 池）在 `2026-07-27-06-ai.md`（单列）实现。

**技术栈：** Kotlin 2.3.10、Spring Boot 4.1、`spring-boot-starter-data-redis`（Lettuce）、
`software.amazon.awssdk:s3` + `s3-request-presigner`、Spring Data MongoDB。

**环境约束：** 无 Docker → Redis/Mongo 集成测试与 bootRun 冒烟本环境跑不了；限流/keystore 等
**纯逻辑用 mock Redis 单元测试**，落库用 Testcontainers 集成测试（照写，CI 跑）。**不要自动 commit。**

```kotlin
package com.ifmix.api.core.common.ratelimit

import org.springframework.data.redis.core.StringRedisTemplate
import org.springframework.stereotype.Component
import java.time.Duration
import java.time.LocalDate
import java.time.ZoneOffset
import java.time.ZonedDateTime

data class RateLimitResult(val allowed: Boolean, val retryAfter: Long? = null)

/** UTC 自然日固定窗口。key=rl:{appId}:{subject}:{yyyy-MM-dd(UTC)}。 */
@Component
class RateLimiter(private val redis: StringRedisTemplate) {

    fun check(appId: String, subject: String, limit: Int): RateLimitResult {
        val key = keyOf(appId, subject)
        val ops = redis.opsForValue()
        val ttl = secondsUntilNextUtcDay()
        ops.setIfAbsent(key, "0", Duration.ofSeconds(ttl))
        val count = ops.increment(key) ?: 1L
        return if (count > limit) RateLimitResult(false, ttl) else RateLimitResult(true)
    }

    fun refund(appId: String, subject: String) {
        redis.opsForValue().decrement(keyOf(appId, subject))
    }

    private fun keyOf(appId: String, subject: String): String {
        val day = LocalDate.now(ZoneOffset.UTC)
        return "rl:$appId:$subject:$day"
    }

    private fun secondsUntilNextUtcDay(): Long {
        val now = ZonedDateTime.now(ZoneOffset.UTC)
        val nextMidnight = now.toLocalDate().plusDays(1).atStartOfDay(ZoneOffset.UTC)
        return Duration.between(now, nextMidnight).seconds.coerceAtLeast(1)
    }
}
```

```kotlin
package com.ifmix.api.core.common.http

import jakarta.servlet.http.HttpServletRequest
import org.springframework.beans.factory.annotation.Value
import org.springframework.stereotype.Component

/** 可信代理 IP 解析。TRUSTED_PROXIES 逗号分隔；仅在来源可信时采信 XFF。 */
@Component
class ClientIpResolver(
    @param:Value("\${app.trusted-proxies:}") trustedProxiesCsv: String,
) {
    private val trusted: Set<String> = trustedProxiesCsv.split(",").map { it.trim() }.filter { it.isNotEmpty() }.toSet()

    fun resolve(request: HttpServletRequest): String {
        request.getHeader("cf-connecting-ip")?.takeIf { it.isNotBlank() }?.let { return it.trim() }
        val remote = request.remoteAddr ?: "unknown"
        if (remote in trusted) {
            val xff = request.getHeader("x-forwarded-for")
            if (!xff.isNullOrBlank()) return xff.split(",").first().trim()
        }
        return remote
    }
}

// S3ObjectStorage.kt
package com.ifmix.api.core.common.storage

import org.springframework.beans.factory.annotation.Value
import org.springframework.stereotype.Component
import software.amazon.awssdk.services.s3.model.GetObjectRequest
import software.amazon.awssdk.services.s3.model.PutObjectRequest
import software.amazon.awssdk.services.s3.presigner.S3Presigner
import software.amazon.awssdk.services.s3.presigner.model.GetObjectPresignRequest
import software.amazon.awssdk.services.s3.presigner.model.PutObjectPresignRequest
import java.time.Duration

@Component
class S3ObjectStorage(
    private val presigner: S3Presigner,
    @param:Value("\${app.storage.bucket:}") private val bucket: String,
) : ObjectStorage {

    override fun presignGetUrl(key: String, ttlSeconds: Long): String {
        val req = GetObjectRequest.builder().bucket(bucket).key(key).build()
        val presigned = presigner.presignGetObject(
            GetObjectPresignRequest.builder().signatureDuration(Duration.ofSeconds(ttlSeconds)).getObjectRequest(req).build(),
        )
        return presigned.url().toString()
    }

    override fun presignPutUrl(key: String, contentType: String, ttlSeconds: Long): String {
        val req = PutObjectRequest.builder().bucket(bucket).key(key).contentType(contentType).build()
        val presigned = presigner.presignPutObject(
            PutObjectPresignRequest.builder().signatureDuration(Duration.ofSeconds(ttlSeconds)).putObjectRequest(req).build(),
        )
        return presigned.url().toString()
    }
}

 * 扫描编排（组合非继承）。newScan：限流 → R2 取图 → AI 扫描 → 落库 → 失败退还额度。
 */
class AntiqueService(
    private val scanCrud: CRUDService<ScanRecordDocument>,
    private val rateLimiter: RateLimiter,
    private val tierResolver: TierResolver,
    private val rateLimitConfig: RateLimitConfig,
    private val storage: ObjectStorage,
    private val scanRunner: ScanRunner,
) {

    fun newScan(ctx: RequestContext, req: NewScanReq, ip: String?): NewScanRes {
        if (ctx.installId.isNullOrBlank()) throw ApiError(ErrorCode.INVALID_REQUEST, "x-install-id: required")
        val subject = resolveSubject(ctx, ip)
        val tier = tierResolver.resolve(ctx)
        val limit = rateLimitConfig.limitForTier(tier)

        val rl = rateLimiter.check(ctx.appId, subject, limit)
        if (!rl.allowed) throw ApiError(ErrorCode.RATE_LIMITED, "rate limited", mapOf("retryAfter" to rl.retryAfter))

        try {
            val imageUrl = storage.presignGetUrl(req.imageKey!!)
            val out = scanRunner.run(ctx, ScanInput(imageUrl, ctx.lang, ctx.currency))
            val doc = ScanRecordDocument().apply {
                installId = ctx.installId
                this.ip = ip
                userId = ctx.userId
                imageKey = req.imageKey
                language = ctx.lang
                currency = ctx.currency
                aiResult = out.result
                modelUsed = out.modelUsed
                isAntique = out.result.isAntique
                name = out.result.name
            }
            val id = scanCrud.createOne(ctx, doc)
            return NewScanRes(id, out.result, out.modelUsed)
        } catch (e: ApiError) {
            rateLimiter.refund(ctx.appId, subject)
            throw e
        } catch (e: Exception) {
            rateLimiter.refund(ctx.appId, subject)
            throw ApiError(ErrorCode.AI_UNAVAILABLE, e.message ?: "ai unavailable")
        }
    }

    fun findById(ctx: RequestContext, id: String): ScanRecordDocument = scanCrud.getById(ctx, id)

    fun listByCursor(ctx: RequestContext, input: CursorQueryInput): Page<ScanRecordDocument> =
        scanCrud.findByCursor(ctx, input)
}
```

> **归属过滤（listScans/getById 的 userId-OR-installId）：** v1 `scan_record` 用通用
> `CRUDService`/`CRUDRepository`，列表默认只按 appId 分片过滤。按 installId/userId 的归属过滤是
> antique 特有逻辑，需在 `scanCrud` 底层 repo 覆写 `extraCriteria`/`extraIdCriteria` 注入
> `(userId == ctx.userId) OR (installId == ctx.installId)`（登录时）或 `installId == ctx.installId`
> （匿名）。因此为 scan_record 建一个 `ScanRecordRepository : CRUDRepository<ScanRecordDocument>`
> 覆写这两个钩子（见任务 9 装配），而非直接用裸 `CRUDRepository`。

package com.ifmix.api.core.modules.antique

import com.ifmix.api.core.common.db.CRUDRepository
import com.ifmix.api.core.common.http.RequestContext
import org.springframework.data.mongodb.core.MongoTemplate
import org.springframework.data.mongodb.core.query.Criteria

/** scan_record 仓储：在 appId 分片过滤之上，附加 userId-OR-installId 归属过滤。 */
class ScanRecordRepository(mongo: MongoTemplate) :
    CRUDRepository<ScanRecordDocument>(mongo, ScanRecordDocument::class.java) {

    private fun ownership(ctx: RequestContext): Criteria {
        val install = Criteria.where("installId").`is`(ctx.installId)
        return if (ctx.userId != null) {
            Criteria().orOperator(Criteria.where("userId").`is`(ctx.userId), install)
        } else {
            install
        }
    }

    override fun extraCriteria(ctx: RequestContext): Criteria = ownership(ctx)
    override fun extraIdCriteria(ctx: RequestContext): Criteria = ownership(ctx)
}
